\noindent\colorbox{lightgray}{%
  \parbox{\linewidth}{%
    \vspace{4pt}
    \textbf{\color{konstanzblue}Thesis-specific rules (CLAUDE.md):}
    \begin{itemize}\setlength\itemsep{1pt}
      \item Prefer \textbf{since} / \textbf{given that} over \textit{because} for causal clauses
      \item Prefer \textbf{consequently} / \textbf{meaning that} over \textit{so} for results
      \item Use \textbf{however} to pivot to a limitation; use \textbf{furthermore} to add a supporting point
      \item Never use: \textit{owing to the fact that}, \textit{it is evident that}, \textit{it should be noted that}
      \item \textbf{Due to} is semi-formal — prefer \textbf{given} or \textbf{since} where possible
    \end{itemize}
